### 🎮 互动游戏：《程序世界大配对》

> **游戏说明**：本游戏包含两个关卡。  
> - **关卡1：概念连连看**——将左侧术语与右侧正确解释连线配对。  
> - **关卡2：生活找程序**——从生活场景图中拖拽出“含有计算机程序”的设备图标到指定区域。  
> 答错会显示解析，答对可进入下一关！支持重新开始，题目顺序随机。

此互动游戏支持在支持 HTML 渲染的环境中直接交互体验。如果未显示游戏，请将下方完整代码保存为 `.html` 文件后用浏览器打开。

<head>

        * { margin: 0; padding: 0; box-sizing: border-box; font-family: "Microsoft YaHei", sans-serif; }

        }

            background: white;

        }
        header {
            background: #1976d2;
            color: white;

        }
        h1 { font-size: 28px; margin-bottom: 8px; }

        .game-area {

        }

            display: flex;
            justify-content: center;
            gap: 20px;

        }
        .tab-btn {

            font-size: 16px;

            border-radius: 20px;
            background: #e3f2fd;

        }
        .tab-btn.active {
            background: #1976d2;
            color: white;
        }
        .level {
            display: none;
            padding: 15px;
            border: 2px dashed #90caf9;
            border-radius: 10px;
            background: #f5fbff;
        }
        .level.active {
            display: block;
        }
        .instructions {
            margin-bottom: 15px;
            color: #0d47a1;
            font-weight: bold;
        }
        /* 关卡1样式 */
        .pairs-container {
            display: grid;
            grid-template-columns: repeat(auto-fill, minmax(200px, 1fr));
            gap: 15px;
            margin-top: 15px;
        }
        .term, .definition {
            padding: 12px;
            border-radius: 8px;
            text-align: center;

            background: #ffe0b2;

<body>
    <div class="container">
        <header>
            <h1>程序世界大配对</h1>
            <p class="subtitle">—— 计算机程序与程序设计语言互动游戏 ——</p>
        </header>
        <div class="game-area">
            <div class="level-tabs">
                <button class="tab-btn active" onclick="switchLevel(1)">关卡1：概念配对</button>

            <!-- 关卡1 -->

                <p class="instructions">🎯 将左侧的【术语】与右侧正确的【解释】配对！点击两个卡片完成连线。</p>
                <div class="pairs-container" id="pairsContainer1"></div>

                <button class="btn reset-btn" onclick="resetGame()">🔄 重新开始</button>

                <div class="scene" id="scene"></div>

        </div>

    <script>
        // 游戏数据：基于教材知识点
        const level1Data = [
            { term: "计算机程序", definition: "计算机可识别运行的指令集合" },
            { term: "CPU", definition: "运算器和控制器的合称" },
            { term: "高级语言", definition: "如Python、C++，接近人类语言" },
            { term: "机器语言", definition: "由0和1组成的二进制代码" },
            { term: "存储程序", definition: "程序和数据事先存入存储器" }
        ];

        // 设备数据：含程序 vs 不含程序
        const devices = [
            { id: 'bell', icon: '🔔', name: '上课铃声', hasProgram: true },
            { id: 'card', icon: '💳', name: '食堂餐卡机', hasProgram: true },
            { id: 'cashier', icon: '🧾', name: '超市收银机', hasProgram: true },
            { id: 'microwave', icon: '🍲', name: '微波炉', hasProgram: true },
            { id: 'atm', icon: '🏧', name: '自动取款机', hasProgram: true },
            { id: 'book', icon: '📚', name: '纸质书', hasProgram: false },
            { id: 'pen', icon: '✏️', name: '钢笔', hasProgram: false },
            { id: 'chair', icon: '🪑', name: '椅子', hasProgram: false }
        ];

        let currentLevel = 1;
        let selectedTerm = null;
        let selectedDef = null;
        let matchedCount = 0;
        let draggedDevice = null;

        // 初始化游戏
        function initGame() {
            initLevel1();
            initLevel2();
        }

        function initLevel1() {
            const container = document.getElementById('pairsContainer1');
            container.innerHTML = '';
            const items = [];

            // 添加术语
            level1Data.forEach(item => {
                items.push({ type: 'term', text: item.term, key: item.term });
            });
            // 添加解释（打乱顺序）
            const shuffledDefs = [...level1Data].sort(() => Math.random() - 0.5);
            shuffledDefs.forEach(item => {
                items.push({ type: 'def', text: item.definition, key: item.term });
            });

            // 再次打乱所有卡片
            const allItems = [...items].sort(() => Math.random() - 0.5);
            allItems.forEach(item => {
                const div = document.createElement('div');
                div.className = item.type === 'term' ? 'term' : 'definition';
                div.textContent = item.text;
                div.dataset.key = item.key;
                div.dataset.type = item.type;
                div.onclick = () => handleCardClick(div);
                container.appendChild(div);
            });
        }

        function handleCardClick(card) {
            if (card.classList.contains('matched')) return;
            const feedback = document.getElementById('feedback1');

            if (card.dataset.type === 'term') {
                if (selectedDef) {
                    feedback.textContent = '请先点击一个术语！';
                    feedback.className = 'feedback incorrect';
                    return;
                }
                if (selectedTerm) selectedTerm.style.border = '2px solid #4caf50';
                selectedTerm = card;
                card.style.border = '3px solid #1976d2';
            } else {
                if (selectedTerm) {
                    selectedDef = card;
                    checkMatch();
                } else {
                    feedback.textContent = '请先点击一个术语！';
                    feedback.className = 'feedback incorrect';
                }
            }
        }

        function checkMatch() {
            const feedback = document.getElementById('feedback1');
            if (selectedTerm.dataset.key === selectedDef.dataset.key) {
                selectedTerm.classList.add('matched');
                selectedDef.classList.add('matched');
                matchedCount++;
                feedback.textContent = '✅ 配对成功！';
                feedback.className = 'feedback correct';
                if (matchedCount === level1Data.length) {
                    setTimeout(() => {
                        feedback.innerHTML = '🎉 关卡1全部完成！点击“关卡2”继续挑战吧！';
                    }, 500);
                }
            } else {
                feedback.textContent = '❌ 配对错误！再想想~';
                feedback.className = 'feedback incorrect';
                selectedTerm.style.border = '2px solid #4caf50';
                selectedDef.style.border = '2px solid #ff9800';
            }
            selectedTerm = null;
            selectedDef = null;
        }

        function initLevel2() {
            const scene = document.getElementById('scene');
            scene.innerHTML = '';
            const dropZone = document.getElementById('dropZone');
            dropZone.innerHTML = '';

            // 随机选择6个设备展示
            const selectedDevices = [...devices].sort(() => Math.random() - 0.5).slice(0, 6);
            selectedDevices.forEach((device, i) => {
                const el = document.createElement('div');
                el.className = 'device';
                el.id = device.id;
                el.innerHTML = device.icon;
                el.title = device.name;
                el.draggable = true;
                el.ondragstart = (e) => {
                    e.dataTransfer.setData('text/plain', device.id);
                    draggedDevice = device;
                };
                // 随机位置
                const left = 30 + (i % 3) * 200;
                const top = 30 + Math.floor(i / 3) * 80;
                el.style.left = `${left}px`;
                el.style.top = `${top}px`;
                scene.appendChild(el);
            });
        }

        // 拖拽事件
        document.getElementById('dropZone').ondrop = (e) => {
            e.preventDefault();
            const id = e.dataTransfer.getData('text/plain');
            const el = document.getElementById(id);
            if (el && !el.classList.contains('dropped')) {
                const feedback = document.getElementById('feedback2');
                const device = devices.find(d => d.id === id);
                if (device.hasProgram) {
                    el.classList.add('dropped');
                    feedback.textContent = `✅ ${device.name} 正确！它内部有程序控制。`;
                    feedback.className = 'feedback correct';
                } else {
                    feedback.textContent = `❌ ${device.name} 错误！它不含计算机程序。`;
                    feedback.className = 'feedback incorrect';
                }
                // 克隆到drop区
                const clone = el.cloneNode(true);
                clone.style.position = 'relative';
                clone.style.margin = '5px';
                clone.draggable = false;
                document.getElementById('dropZone').appendChild(clone);
            }
        };

        document.getElementById('dropZone').ondragover = (e) => {
            e.preventDefault();
        };

        function switchLevel(level) {
            currentLevel = level;
            document.querySelectorAll('.level').forEach(l => l.classList.remove('active'));
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            document.getElementById(`level${level}`).classList.add('active');
            document.querySelector(`.tab-btn:nth-child(${level})`).classList.add('active');
        }

        function resetGame() {
            matchedCount = 0;
            selectedTerm = null;
            selectedDef = null;
            draggedDevice = null;
            document.getElementById('feedback1').textContent = '';
            document.getElementById('feedback2').textContent = '';
            if (currentLevel === 1) {
                initLevel1();
            } else {
                initLevel2();
            }
        }

        // 启动游戏
        window.onload = initGame;
    </script>
</body>
</html>
